\subsection{Domínio}

\begin{itemize}
	\item Blocos $\mathcal{B}=\{a,b,c,d\}$ com comprimento $\ell$:
	$\ell(a)=\ell(b)=1$, $\ell(c)=2$, $\ell(d)=3$.
	\item Pontos do eixo $\mathcal{X}=\{0,\dots,6\}$ e slots $s_i=[i,i+1]$,
	$i\in\{0,\dots,5\}$ (6 slots, não 7).
	\item Níveis $\mathcal{L}=\{0,1,2,3\}$ ($0$ = sobre a mesa) e instantes $t\in\{0,\dots,H\}$.
	\item Um bloco $b$ que começa no ponto $p$ ocupa
	$\spn(b,p)=\{p,\dots,p+\ell(b)-1\}$, com $p\in\{0,\dots,6-\ell(b)\}$.
	\item $\ov(b,p,y,q) \leftrightarrow \spn(b,p)\cap\spn(y,q)\neq\emptyset$.
\end{itemize}

\subsection{Predicados e axiomas}
%\afazer{Tabela dos predicados ($\at$, $\lev$, $\cov$, $\livre$, $\esp$, $\clr$, $\on$) e os axiomas em LPO (unicidade, exclusão, estabilidade etc.).}

\begin{center}
	\begin{tabular}{@{}lll@{}}
		\toprule
		Predicado & Significado & Tipo\\
		\midrule
		$\at(b,p,t)$ & $b$ começa no ponto $p$ no instante $t$ & fluente\\
		$\lev(b,l,t)$ & $b$ está no nível $l$ & fluente\\
		$\cov(b,s,l,t)$ & $b$ cobre o slot $s$ no nível $l$ & derivado\\
		$\livre(s,l,t)$ & nenhum bloco cobre $s$ no nível $l$ & derivado\\
		$\esp(b,s,t)$ & \iffalse \textbf{(novo)}\fi há espaço livre em cima de $b$, no slot $s$ & derivado\\
		$\clr(b,t)$ & não há nada em cima de $b$ (espaço livre em todos os slots) & derivado\\
		$\on(b,y,t)$ & $b$ está apoiado em $y\in\mathcal{B}\cup\{\mesaT\}$ & derivado\\
		\bottomrule
	\end{tabular}
\end{center}

\noindent Para descrever um estado,basta dizer onde cada bloco começa $\at$ e em que nível ele está $\lev$; todo o resto é
definido a partir deles pelos axiomas abaixo (para todo $b,y,p,s,l,t$):
\begin{align}
	&\forall b,t\;\exists!\,p\;\at(b,p,t) \qquad \forall b,t\;\exists!\,l\;\lev(b,l,t)
	\tag{A1 unicidade}\\
	&\cov(b,s,l,t) \leftrightarrow \lev(b,l,t)\land\exists p\,
	\bigl(\at(b,p,t)\land s\in\spn(b,p)\bigr) \tag{A2}\\
	&b\neq b' \rightarrow \lnot\bigl(\cov(b,s,l,t)\land\cov(b',s,l,t)\bigr)
	\tag{A3 exclusão}\\
	&\livre(s,l,t) \leftrightarrow \lnot\exists b\;\cov(b,s,l,t) \tag{A4}\\
	&\esp(b,s,t) \leftrightarrow \exists l\,\bigl(\cov(b,s,l,t)\land
	\livre(s,l+1,t)\bigr) \tag{A5a espaço}\\
	&\clr(b,t) \leftrightarrow \forall s,l\,\bigl(\cov(b,s,l,t)\rightarrow
	\esp(b,s,t)\bigr) \tag{A5b clear}\\
	&\on(b,\mesaT,t) \leftrightarrow \lev(b,0,t) \tag{A6}\\
	&\on(b,y,t) \leftrightarrow \exists p,q,l\;\at(b,p,t)\land\at(y,q,t)\land
	\lev(b,l,t)\land\lev(y,l-1,t)\land\ov(b,p,y,q) \notag\\
	&\at(b,p,t)\land\lev(b,l,t)\land l>0 \rightarrow
	\Bigl|\bigl\{s\in\spn(b,p):\lnot\livre(s,l-1,t)\bigr\}\Bigr|
	\ge\Bigl\lceil\tfrac{\ell(b)}{2}\Bigr\rceil \tag{A7 estabilidade}
\end{align}

O predicado $\esp$ é novo. Ele funciona como um $\clr$ olhado slot por slot: diz se
há espaço livre em cima do bloco $b$ naquele slot. O $\clr$ continua existindo: um
bloco está livre quando todos os seus slots têm espaço livre em cima (A5b). No último
nível ($l=3$) não existe nível acima, então consideramos que o espaço em cima está
sempre livre.

A regra A7 é a do equilíbrio: um bloco empilhado precisa ter pelo menos metade do
seu comprimento apoiada. Os blocos $a$ e $b$ precisam de 1 slot de apoio, o $c$
também de 1 (pode ficar com metade para fora) e o $d$ de 2. É por isso que a ponte
de $S_0$ é permitida: o $d$ fica sobre $a$ e $b$, com o slot $s_4$ vazio embaixo.


\subsection{A ação \texorpdfstring{$\mv(b,y,p)$}{move(b,y,p)}}
\iffalse \afazer{Pré-condições (P1, P2, \dots) e efeitos em estilo STRIPS (ADD / DEL).}
\fi

``Mover o bloco $b$ para cima de $y$ (ou da mesa $\mesaT$), começando no ponto $p$.''
Seja $(p_0,l_0)$ a posição atual de $b$ e $l^\ast$ o nível de chegada:
$l^\ast=0$ se $y=\mesaT$ e $l^\ast=\lev(y)+1$ se $y$ é bloco.

\paragraph{Pré-condições.}
\begin{enumerate}
	\item[P1] $\clr(b)$: nada em cima de $b$ (só se move bloco livre).
	\item[P2] se $y$ é bloco: $\ov(b,p,y,q)$, onde $\at(y,q)$ ($b$ fica de fato sobre $y$),
	e $\esp(y,s)$ para todo $s\in\spn(b,p)\cap\spn(y,q)$: há espaço livre em cima
	de $y$ nos slots onde $b$ vai pousar (sem contar o próprio $b$, se ele já
	estiver ali). %\textbf{Substitui o $\clr(y)$ do manual.}
	\item[P3] $(p,l^\ast)\neq(p_0,l_0)$ (não é um \emph{no-op}).
	\item[P4] $\forall s\in\spn(b,p)$: $s$ está livre no nível $l^\ast$, sem contar o próprio $b$.
	\item[P5] \textbf{sombra}: $\forall s\in\spn(b,p),\ \forall l'>l^\ast$: $s$ está livre
	no nível $l'$ (não se enfia um bloco embaixo de outro).
	\item[P6] $l^\ast\le 3$.
\end{enumerate}
A estabilidade (A7) não precisa ser pré-condição: ela é uma restrição que vale
em \emph{todo} estado, então o estado de chegada tem que respeitá-la.

\paragraph{Efeitos.}
Depois de $\mv(b,y,p)$, o bloco $b$ sai de $(p_0,l_0)$ e passa a estar em $(p,l^\ast)$.
A tabela mostra o que passa a valer (ADD) e o que deixa de valer (DEL), no estilo STRIPS.
Chamamos de $z$ o bloco que apoiava $b$ antes e de $y'$ o bloco que apoia $b$ depois.

\begin{description}
	\item[ADD] $\at(b,p)$, $\lev(b,l^\ast)$, $\cov(b,s,l^\ast)$ para $s\in\spn(b,p)$,
	$\on(b,y')$ para cada novo apoio $y'$ (um ou dois, se for ponte),
	$\esp(z,s)$ para cada antigo apoio $z$, nos slots que $b$ deixou;
	$\clr(z)$ se $z$ ficou sem nada em cima;
	$\livre(s,l_0)$ para os slots que $b$ deixou.
	\item[DEL] $\at(b,p_0)$, $\lev(b,l_0)$, $\cov(b,s,l_0)$ antigos,
	$\on(b,z)$ antigos, $\esp(y',s)$ e $\clr(y')$ para cada novo apoio $y'$
	(nos slots onde $b$ pousou),
	$\livre(s,l^\ast)$ para os slots que $b$ passou a ocupar.
\end{description}

Se o bloco só muda de posição e continua no mesmo nível (ou o contrário), o que
não mudou não entra na lista.

\emph{Exemplo:} $\mv(d,c,0)$ em $S_0$ tira o $d$ de cima de $a$ e $b$ e o coloca
sobre $c$. Passam a valer $\at(d,0)$, $\on(d,c)$, $\clr(a)$ e $\clr(b)$. Deixam de
valer $\at(d,3)$, $\on(d,a)$, $\on(d,b)$ e $\clr(c)$. O nível do $d$ continua 1,
então $\lev$ não muda.
